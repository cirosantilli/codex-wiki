"""Original 2021 Particle and Nuclear Physics solution figure.

Writes the PNG basename to the current working directory. Dependencies are
numpy and matplotlib, declared in the repository pyproject.toml.
Tested with Python 3.14.4, NumPy 2.3.5, Matplotlib 3.10.7.
"""
from pathlib import Path
import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt
from matplotlib.patches import Rectangle
import numpy as np


def line(ax, a, b, kind="fermion", backwards=False, label=None):
    a, b = np.array(a, dtype=float), np.array(b, dtype=float)
    d = b - a
    n = np.array([-d[1], d[0]]) / np.linalg.norm(d)
    if kind in ("wavy", "gluon"):
        t = np.linspace(0, 1, 300)
        phase = 16 * np.pi * t
        curve = a[:, None] + d[:, None] * t + 0.045 * n[:, None] * np.sin(phase)
        if kind == "gluon":
            curve += 0.045 * (d / np.linalg.norm(d))[:, None] * (np.cos(phase)-1) * np.sin(np.pi*t)
        ax.plot(*curve, color="#27618c", lw=1.6)
    else:
        ax.plot([a[0], b[0]], [a[1], b[1]], color="#333333", lw=1.5)
        if kind == "fermion":
            if backwards:
                a, b = b, a
            ax.annotate("", xy=a+0.61*(b-a), xytext=a+0.43*(b-a),
                        arrowprops={"arrowstyle":"-|>","color":"#333333","lw":1.3})
    if label:
        pos=(a+b)/2+0.17*n
        ax.text(*pos, label, fontsize=12, ha="center", va="center", color="#174564")


def text(ax, x, y, label, **kwargs):
    ax.text(x, y, label, ha="center", va="center", fontsize=12, **kwargs)


def setup(ax, xlim=(-.3,7), ylim=(-2.1,3.9)):
    ax.set_xlim(*xlim);ax.set_ylim(*ylim);ax.axis("off")


def save(fig):
    fig.savefig(Path(__file__).with_suffix(".png").name,dpi=145,facecolor="white")
    plt.close(fig)


def main():
    fig,axes=plt.subplots(2,1,figsize=(7.6,8.1))
    for ax,three in zip(axes,[False,True]):
        line(ax,(.5,1.1),(2.0,1.1),backwards=True);line(ax,(2.0,1.1),(5.6,1.1),backwards=True)
        line(ax,(.5,0),(5.6,0))
        line(ax,(2,1.1),(3.0,2.4),"wavy",label=r"$W^+$")
        line(ax,(3.0,2.4),(5.6,3.3));line(ax,(3.0,2.4),(5.6,2.4),backwards=True)
        for x,y,l in [(.18,1.1,r"$\bar s$"),(.18,0,r"$u$"),(5.95,3.3,r"$u$"),(5.95,2.4,r"$\bar d$"),(5.95,1.1,r"$\bar u$"),(5.95,0,r"$u$")]:text(ax,x,y,l)
        if three:
            line(ax,(1.5,0),(3.2,-1.15),"gluon",label=r"$g$")
            line(ax,(3.2,-1.15),(5.6,-.85));line(ax,(3.2,-1.15),(5.6,-1.8),backwards=True)
            text(ax,5.95,-.85,r"$d$");text(ax,5.95,-1.8,r"$\bar d$")
        ax.set_title(r"$K^+\to\pi^+\pi^+\pi^-$: extra soft pair creation" if three else r"$K^+\to\pi^+\pi^0$: use the $u\bar u$ neutral-pion component",fontsize=11)
        setup(ax,ylim=(-2.25,3.8))
    fig.suptitle("Representative kaon weak-decay quark flow",fontsize=14)
    fig.text(.5,.015,"Spectator u line below; final quarks combine into the pions specified in the solution.",ha="center",fontsize=9)
    fig.tight_layout(rect=(0,.045,1,.96),h_pad=1.9);save(fig)

if __name__=="__main__":main()
