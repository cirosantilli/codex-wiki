Fix the top-right corner (i,j) and a bottom boundary k. A width w is feasible precisely when every row m∈[i,k) has at least w consecutive true entries ending at column j. Thus the largest feasible width is the minimum of those row run lengths, and the area is that width times k−i.
def maxRectangleAt(i: Int, j: Int): Int = {
  var k = i
  var width = j + 1
  var best = 0
  while (k < N) {
    width = math.min(width, rowCount(k)(j))
    k += 1
    best = math.max(best, (k - i) * width)
  }
  best
}
